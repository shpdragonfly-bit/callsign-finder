function getProductNpayButtonSetting(parent) {
	var gsettings = (parent && parent.gsettings) ? parent.gsettings : {};
	if(typeof gsettings == 'string') {
		try {
			gsettings = JSON.parse(gsettings || '{}');
		} catch(e) {
			gsettings = {};
		}
	}
	return (gsettings && gsettings.npay_button == 'N') ? 'N' : 'Y';
}

function applyProductNpayButtonVisibility() {
	var hideNpayButton = (typeof property != 'undefined' && getProductNpayButtonSetting(property.PARENT) == 'N'),
		styleId = 'npay-button-hidden-style',
		style = document.getElementById(styleId);

	if(hideNpayButton) {
		if(!style) {
			style = document.createElement('style');
			style.id = styleId;
			style.appendChild(document.createTextNode('#naverpay-buy,.naverpay-buy-btn{display:none!important;}'));
			document.head.appendChild(style);
		}
		$('.naverpay-buy-btn').remove();
	} else if(style) {
		style.parentNode.removeChild(style);
	}
}
