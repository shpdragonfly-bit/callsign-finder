<script>
	/* 쿠키 읽기 */
	function getCookie(name) {
		var nameOfCookie = name + "=";
		var x = 0;
		while (x <= document.cookie.length) {
			var y = x + nameOfCookie.length;
			if (document.cookie.substring(x, y) == nameOfCookie) {
				var endOfCookie = document.cookie.indexOf(";", y);
				if (endOfCookie == -1) endOfCookie = document.cookie.length;
				return unescape(document.cookie.substring(y, endOfCookie));
			}
			x = document.cookie.indexOf(" ", x) + 1;
			if (x == 0) break;
		}
		return "";
	}

	/* 오늘 자정 만료 쿠키 */
	function setCookieToday(name) {
		var expire = new Date();
		expire.setHours(23, 59, 59, 0);
		document.cookie = name + "=no; expires=" + expire.toUTCString() + "; path=/";
	}

	/* 닫기 버튼 */
	function closeLayerPopup(id) {
		var chk = document.getElementById('chk_' + id);
		if (chk && chk.checked) {
			setCookieToday(id);
		}
		var el = document.getElementById('popup_' + id);
		if (el) el.style.display = 'none';
	}

	/* 페이지 로드 후 쿠키 확인하여 표시 */
	$(document).ready(function () {

		}
		if (getCookie("20260602") != "no") {
			$('#popup_20260602').css('display', 'flex');
		}
	});
