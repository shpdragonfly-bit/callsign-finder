function get_timetable(_event, _this, line, station) {
  var url = '/timetable/' + line + '/' + station;
  if (md.mobile()) {
    window.location.href = url;
  } else {
    $('li.station a').removeClass('selected');
    $(_this).addClass('selected');

    $('#stationTimeSchedule').attr('src', url).hide();

    $('#stationTimeSchedule').on("load", function() {
      $(this).fadeIn('fast');
      $('.frameWrap').fadeIn('fast');
    });

    _event.preventDefault();
  }
}
